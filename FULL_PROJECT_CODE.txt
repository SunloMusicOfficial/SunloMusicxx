async def ensure_assistant_in_chat(chat_id: int) -> str:
    if not assistant: return "failed"

    try:
        member = await bot.get_chat_member(chat_id, assistant.me.id)
        if member.status in [enums.ChatMemberStatus.BANNED, enums.ChatMemberStatus.RESTRICTED]: return "banned"
        if member.status in [enums.ChatMemberStatus.MEMBER, enums.ChatMemberStatus.ADMINISTRATOR, enums.ChatMemberStatus.OWNER]: return "ok"
    except: pass

    try:
        member = await assistant.get_chat_member(chat_id, "me")
        if member.status in [enums.ChatMemberStatus.BANNED, enums.ChatMemberStatus.RESTRICTED]: return "banned"
        if member.status in [enums.ChatMemberStatus.MEMBER, enums.ChatMemberStatus.ADMINISTRATOR, enums.ChatMemberStatus.OWNER]: return "ok"
    except Exception as e:
        if "USER_BANNED" in str(e).upper(): return "banned"

    try:
        chat = await bot.get_chat(chat_id)
        if chat.username:
            try:
                await assistant.join_chat(chat.username)
                return "ok"
            except UserAlreadyParticipant: return "ok"
            except Exception as e:
                if "USER_BANNED" in str(e).upper(): return "banned"
    except: pass

    try:
        try: link = await bot.export_chat_invite_link(chat_id)
        except Exception as e:
            if "CHAT_ADMIN_REQUIRED" in str(e).upper(): return "failed_no_admin"
            raise e
            
        await asyncio.sleep(1)
        try: await assistant.join_chat(link)
        except UserAlreadyParticipant: return "ok"
        except Exception as e:
            if "USER_BANNED" in str(e).upper(): return "banned"
            
        await asyncio.sleep(2) 
        try:
            member = await assistant.get_chat_member(chat_id, "me")
            if member.status in [enums.ChatMemberStatus.BANNED, enums.ChatMemberStatus.RESTRICTED]: return "banned"
            return "ok"
        except:
            try:
                me_ass = await assistant.get_me()
                await bot.approve_chat_join_request(chat_id, me_ass.id)
                return "ok"
            except: pass
        return "ok" 
    except Exception as e:
        if "USER_BANNED" in str(e).upper(): return "banned"

    return "failed"

def cleanup_file(url: str):
    if url and not url.startswith("http") and os.path.exists(url) and url.endswith(('.mp4', '.webm', '.m4a', '.mp3')):
        try: os.remove(url)

def _build_audio_filters(chat_id: int, effect: str = "none", seek_time: int = 0, duration: int = 0) -> str | None:
    ctrl = AUDIO_CONTROLS.get(chat_id, {})
    speed = ctrl.get("speed", 1.0)
    volume = ctrl.get("volume", 1.0)
    eq = ctrl.get("eq", "none")
    af_filters = []
    if chat_id in CROSSFADE_CHATS and seek_time == 0:
        af_filters.append("afade=t=in:st=0:d=5")
    if chat_id in CROSSFADE_CHATS and duration > 10 and seek_time == 0:
        af_filters.append(f"afade=t=out:st={duration-5}:d=5")
    if effect == "bass": af_filters.append("bass=g=10")
    elif effect == "8d": af_filters.append("apulsator=hz=0.1")
    elif effect == "nightcore": af_filters.append("atempo=1.25,asetrate=44100*1.25")
    elif effect == "chipmunk": af_filters.append("asetrate=44100*1.5,atempo=1/1.5")
    elif effect == "monster": af_filters.append("asetrate=44100*0.7,atempo=1/0.7")
    else:
        if eq != "none":
            eq_map = {"bass": "bass=g=12", "pop": "equalizer=f=1000:width_type=h:width=200:g=5", "rock": "equalizer=f=100:width_type=h:width=200:g=3,equalizer=f=3000:width_type=h:width=200:g=4", "jazz": "equalizer=f=200:width_type=h:width=200:g=2", "vocal": "equalizer=f=2000:width_type=h:width=1000:g=5"}
            if eq in eq_map: af_filters.append(eq_map[eq])
    if speed != 1.0: af_filters.append(f"atempo={speed}")
    if volume != 1.0: af_filters.append(f"volume={volume}")
    params = []
    if seek_time > 0: params.append(f"-ss {seek_time}")
    if af_filters: params.append(f"-af \"{','.join(af_filters)}\"")
    return " ".join(params) if params else None

async def check_empty_vc(chat_id):
    try:
        members = await bot_module.bot.get_chat_members(chat_id, filter=enums.ChatMembersFilter.VOICE_CHAT)
        humans = [m for m in members if not m.user.is_bot]
        if len(humans) == 0:
            if chat_id not in EMPTY_VC_TIMERS: EMPTY_VC_TIMERS[chat_id] = time.time()
            elif time.time() - EMPTY_VC_TIMERS[chat_id] > 180:
                await bot_module.bot.send_message(chat_id, "💤 <b>No humans in VC. Leaving...</b>")
                await stop_stream(chat_id)
                from SunloMusic.queues import clear_queue
                clear_queue(chat_id)
                if chat_id in EMPTY_VC_TIMERS: del EMPTY_VC_TIMERS[chat_id]
                return True
        else:
            if chat_id in EMPTY_VC_TIMERS: del EMPTY_VC_TIMERS[chat_id]
    except: pass
    return False

async def stream_end_handler(client, update: StreamEnded):
    from SunloMusic.queues import get_queue
    chat_id = update.chat_id
    q = get_queue(chat_id)
    if getattr(q, "is_playing_tts", False):
        q.is_playing_tts = False
        return
    if not q.now_playing: return
    if q.loop == 1 and q.now_playing:
        await play_stream(chat_id, q.now_playing.url, q.now_playing.video, duration=q.now_playing.duration)
        q.start_time = time.time(); q.total_paused = 0; q.paused = False
        return
    if q.loop == 2 and q.now_playing: q.queue.append(q.now_playing)
    await start_next_song(chat_id, q)

async def play_stream(chat_id: int, url: str, video: bool = False, effect: str = "none", seek_time: int = 0, duration: int = 0):
    if not call: return False
    if not url.startswith("http") and not os.path.exists(url):
        LOGGER.error(f"❌ File not found: {url}")
        return False
    if video and (url.endswith('.m4a') or url.endswith('.mp3')): video = False
    ffmpeg_args = _build_audio_filters(chat_id, effect, seek_time, duration)
    if video: stream = MediaStream(media_path=url, audio_parameters=AudioQuality.HIGH, video_parameters=VideoQuality.HD_720p, ffmpeg_parameters=ffmpeg_args)
    else: stream = MediaStream(media_path=url, audio_parameters=AudioQuality.HIGH, video_flags=MediaStream.Flags.IGNORE, ffmpeg_parameters=ffmpeg_args)
    try:
        try: await bot_module.assistant.get_chat(chat_id)
        except: pass
        await asyncio.wait_for(call.play(chat_id, stream, config=GroupCallConfig(auto_start=True)), timeout=120.0) 
        return True
    except Exception as e:
        err_str = str(e)
        LOGGER.error(f"Play error: {err_str}")
        if "CHANNEL_INVALID" in err_str or "USER_BANNED" in err_str or "CHAT_ADMIN_REQUIRED" in err_str:
            return False # ✅ CRITICAL: Stop the card from being sent
        if "INTERDC_X_CALL_ERROR" in err_str:
            await asyncio.sleep(5)
            try:
                await asyncio.wait_for(call.play(chat_id, stream, config=GroupCallConfig(auto_start=True)), timeout=120.0)
                return True
            except: pass
        return True

from functools import wraps
from pyrogram import enums
from pyrogram.types import Message, CallbackQuery
from SunloMusic.config import config
from SunloMusic.database import is_sudo, get_chat
from SunloMusic.texts import NO_SUDO_TEXT, NO_ADMIN_TEXT
from SunloMusic.error_handler import log_error_to_group

def catch_errors(func):
    @wraps(func)
    async def wrapper(client, update, *args, **kwargs):
        try: return await func(client, update, *args, **kwargs)
        except Exception as e: await log_error_to_group(update, e)
    return wrapper

def sudo_only(func):
    @wraps(func)
    async def wrapper(client, update, *args, **kwargs):
        user = update.from_user
        if not user: return
        
        # Direct owner/sudo check
        if user.id == config.OWNER_ID or user.id in config.sudo_list:
            return await func(client, update, *args, **kwargs)
            
        try:
            if await is_sudo(user.id): 
                return await func(client, update, *args, **kwargs)
        except: pass
        
        if isinstance(update, Message): await update.reply_text(NO_SUDO_TEXT)
        elif isinstance(update, CallbackQuery): await update.answer(NO_SUDO_TEXT, show_alert=True)
    return wrapper

def admin_only(func):
    @wraps(func)
    async def wrapper(client, message: Message, *args, **kwargs):
        if not message.from_user: return
        
        if message.from_user.id == config.OWNER_ID or message.from_user.id in config.sudo_list:
            return await func(client, message, *args, **kwargs)
            
        try:
            if await is_sudo(message.from_user.id): 
                return await func(client, message, *args, **kwargs)
        except: pass
            
        if message.chat.type != enums.ChatType.PRIVATE:
            try:
                member = await message.chat.get_member(message.from_user.id)
                if member.privileges:
                    can_manage = getattr(member.privileges, "can_manage_video_chats", False) or \
                                 getattr(member.privileges, "can_manage_voice_chats", False) or \
                                 member.privileges.can_manage_chat
                    if can_manage: return await func(client, message, *args, **kwargs)
            except Exception: pass
        await message.reply_text(NO_ADMIN_TEXT)
    return wrapper

import traceback
from pyrogram.types import Message, CallbackQuery
from SunloMusic.config import config
from SunloMusic.logger import LOGGER

async def log_error_to_group(update, error: Exception):
    error_msg = traceback.format_exc()
    LOGGER.error(error_msg)
    
    user = update.from_user if hasattr(update, 'from_user') else None
    user_mention = user.mention if user else "Unknown"
    user_id = user.id if user else 0
    
    if isinstance(update, Message):
        chat_title = update.chat.title or update.chat.first_name
        chat_id = update.chat.id
        cmd = update.text or update.caption or "Unknown Command"
    elif isinstance(update, CallbackQuery):
        chat_title = update.message.chat.title or update.message.chat.first_name
        chat_id = update.message.chat.id
        cmd = update.data
    else:
        chat_title = "Unknown"
        chat_id = 0
        cmd = "Unknown"

    text = (
        f"<b>🚨 EMERGENCY ERROR 🚨</b>\n"
        f"━━━━━━━━━━━━━━━━━━━━\n"
        f"🏷 <b>Owner:</b> <a href='tg://user?id={config.OWNER_ID}'>Sakil</a>\n"
        f"👤 <b>User:</b> {user_mention} (<code>{user_id}</code>)\n"
        f"💬 <b>Chat:</b> {chat_title} (<code>{chat_id}</code>)\n"
        f"🛠 <b>Action:</b> <code>{cmd}</code>\n"
        f"━━━━━━━━━━━━━━━━━━━━\n"
        f"<b>Error Details:</b>\n"
        f"<pre>{error_msg[-3000:]}</pre>" # Show last 3000 chars
    )

    try:
        if config.LOGGER_ID:
            from SunloMusic.bot import bot
            if bot:
                await bot.send_message(config.LOGGER_ID, text)
    except Exception as e:
        LOGGER.error(f"Failed to send error log: {e}")

    try:
        user_text = (
            f"🚨 <b>Oops! An unexpected error occurred.</b>\n\n"
            f"Don't worry, the Owner (<a href='tg://user?id={config.OWNER_ID}'>Sakil</a>) has been tagged and "
            f"the error details have been sent to the Log Group automatically."
        )
        if isinstance(update, Message):
            await update.reply_text(user_text)
        elif isinstance(update, CallbackQuery):
            await update.answer("🚨 Error occurred! Owner has been notified.", show_alert=True)
    except: pass

def format_duration(seconds: int) -> str:
    if not seconds: return "00:00"
    h, rem = divmod(int(seconds), 3600)
    m, s = divmod(rem, 60)
    return f"{h:02d}:{m:02d}:{s:02d}" if h else f"{m}:{s:02d}"

def make_progress_bar(current: int, total: int, length: int = 7) -> str:
    if total <= 0: return "─" * length
    pos = int(length * current / total)
    pos = max(0, min(pos, length - 1))
    # Thin line with circle
    return "─" * pos + "○" + "─" * (length - pos - 1) if length > 1 else "○"

async def _background_pinger():
    await asyncio.sleep(60)
    url = os.environ.get("RENDER_EXTERNAL_URL")
    if not url: return
    ping_url = f"{url.rstrip('/')}/ping"
    async with aiohttp.ClientSession() as session:
        while True:
            try:
                async with session.get(ping_url, timeout=10) as resp: await resp.read()
            except: pass
            await asyncio.sleep(600)

async def start_keep_alive():
    app = web.Application()
    app.router.add_get("/", _handle_dashboard)
    app.router.add_get("/ping", _handle_ping)
    app.router.add_get("/movie", _handle_movie_download) 
    port = int(os.environ.get("PORT", 8080))
    runner = web.AppRunner(app)
    await runner.setup()
    site = web.TCPSite(runner, "0.0.0.0", port)
    await site.start()
    LOGGER.info(f"✅ Web Server running on port {port}")
    asyncio.create_task(_background_pinger())

import aiohttp
import re
import asyncio
from urllib.parse import quote
from pyrogram.errors import FloodWait
from SunloMusic.logger import LOGGER

async def _api_get(url, headers, timeout=10):
    try:
        conn = aiohttp.TCPConnector(family=2) # Force IPv4
        async with aiohttp.ClientSession(headers=headers, connector=conn) as s:
            async with s.get(url, timeout=timeout) as r:
                if r.status == 200: return await r.json()
    except Exception as e: LOGGER.error(f"API Error: {e}")
    return None

def _parse_song_info(query: str):
    clean = re.sub(r'[\(\[].*?[\)\]]', '', query)
    parts = [p.strip() for p in re.split(r'[|\-]', clean) if p.strip()]
    if not parts: return query, ""
    title = parts[0]
    artist = parts[1] if len(parts) > 1 else ""
    if " by " in title.lower():
        t_parts = re.split(r' by ', title, flags=re.IGNORECASE)
        title = t_parts[0].strip()
        artist = t_parts[1].strip() if len(t_parts) > 1 else artist
    return title.strip(), artist.strip()

async def fetch_lyrics_text(query: str, duration: int = 0) -> str | None:
    title, artist = _parse_song_info(query)
    headers = {"User-Agent": "SunloMusic/1.0"}
    
    # LRCLIB
    try:
        url = f"https://lrclib.net/api/get?track_name={quote(title)}"
        if artist: url += f"&artist_name={quote(artist)}"
        if duration > 0: url += f"&duration={duration}"
        data = await _api_get(url, headers)
        if data and "plainLyrics" in data and data["plainLyrics"]: return data["plainLyrics"]
        
        search_url = f"https://lrclib.net/api/search?q={quote(title)}"
        if artist: search_url += f"+{quote(artist)}"
        search_data = await _api_get(search_url, headers)
        if search_data and isinstance(search_data, list) and len(search_data) > 0:
            for track in search_data:
                if "plainLyrics" in track and track["plainLyrics"]: return track["plainLyrics"]
    except: pass

    # JioSaavn
    try:
        search_url = f"https://saavnapi-nine.vercel.app/result/?query={quote(title)}"
        data = await _api_get(search_url, headers)
        if data and "songs" in data and "results" in data["songs"] and len(data["songs"]["results"]) > 0:
            lyrics_url = f"https://saavnapi-nine.vercel.app/lyrics/?id={data['songs']['results'][0]['id']}"
            lyrics_data = await _api_get(lyrics_url, headers)
            if lyrics_data and "lyrics" in lyrics_data and lyrics_data["lyrics"]:
                lyrics = re.sub(r'<br\s*/?>', '\n', lyrics_data["lyrics"])
                return re.sub(r'<.*?>', '', lyrics).strip()
    except: pass

    # Lyrics.ovh
    if artist and title:
        try:
            url = f"https://api.lyrics.ovh/v1/{quote(artist)}/{quote(title)}"
            data = await _api_get(url, headers)
            if data and "lyrics" in data and data["lyrics"]: return data["lyrics"]
        except: pass

    return None

async def start_live_karaoke(bot, chat_id, title, duration, q):
    msg = await bot.send_message(chat_id, f"🎤 <b>Fetching live lyrics for:</b> <i>{title}</i>...")
    
    try:
        lyrics = await asyncio.wait_for(fetch_lyrics_text(title, duration), timeout=20.0)
        
        if not lyrics:
            await msg.edit_text(f"🎤 <b>Karaoke:</b> Lyrics not found for <i>{title}</i>.")
            return

        lines = [line.strip() for line in lyrics.split('\n') if line.strip()]
        if not lines: return

        # ✅ Fixed 5 second delay as requested
        delay = 5.0
        header = "🎤 <b>Live Karaoke</b>\n\n<blockquote>"
        accumulated = lines[0]
        
        await msg.edit_text(f"{header}{accumulated}</blockquote>")
        q.karaoke_msg_id = msg.id

        for i in range(1, len(lines)):
            if not q.now_playing or q.now_playing.title != title or q.paused: break
            
            await asyncio.sleep(delay)
            next_line = lines[i]
            
            if len(accumulated) + len(next_line) + len(header) > 4000:
                try: await msg.edit_text(f"{header}{accumulated}</blockquote>")
                except: pass
                accumulated = next_line
                msg = await bot.send_message(chat_id, f"{header}{accumulated}</blockquote>")
                q.karaoke_msg_id = msg.id
            else:
                accumulated += f"\n{next_line}"
                try:
                    await msg.edit_text(f"{header}{accumulated}</blockquote>")
                except FloodWait as e:
                    await asyncio.sleep(e.value)
                    try: await msg.edit_text(f"{header}{accumulated}</blockquote>")
                    except: pass
                except Exception:
                    pass

    except asyncio.TimeoutError:
        await msg.edit_text("❌ <b>Karaoke Error:</b> Timed out.")
    except asyncio.CancelledError: pass
    except Exception as e:
        try: await msg.edit_text(f"❌ <b>Karaoke Error:</b> <code>{e}</code>")
        except: pass

    total_paused: float = 0.0
    playing_message_id: int = 0

    is_playing_tts: bool = False

_queues: dict[int, ChatQueue] = {}

            parts = list(map(int, time_str.split(':')))
            if len(parts) == 2: seconds = parts[0] * 60 + parts[1]
            elif len(parts) == 3: seconds = parts[0] * 3600 + parts[1] * 60 + parts[2]
        else:
            seconds = int(time_str)
    except:
        return await m.reply_text("❌ <b>Invalid time format.</b>")
        
    if seconds >= q.now_playing.duration:
        return await m.reply_text("❌ <b>Seek time is greater than song duration!</b>")

    await play_stream(m.chat.id, q.now_playing.url, q.now_playing.video, seek_time=seconds)
    
    q.start_time = time.time() - seconds
    q.total_paused = 0
    q.paused = False
    
    await status.delete()

def _fmt_size(size: int) -> str:
    for unit in ['B', 'KB', 'MB', 'GB']:
        if size < 1024: return f"{size:.1f}{unit}"
        size /= 1024
    return f"{size:.1f}TB"

async def _do_download(m: Message, video: bool):
    user_id = m.from_user.id
    query = " ".join(m.command[1:])
    
    is_priv = await is_sudo(user_id)
    if not is_priv:
        bal = await get_coins(user_id)
        if bal < DOWNLOAD_COST:
            return await m.reply_text(
                f"❌ <b>Not enough coins!</b>\n\n"
                f"💰 Your balance: <code>{bal}</code>\n"
                f"💵 Needed: <code>{DOWNLOAD_COST}</code>\n\n"
                f"Use /daily to claim coins!"
            )
    
    status = await m.reply_text(f"🔍 <b>Searching & downloading {'video' if video else 'audio'}...</b>\nThis may take a minute.")
    
    try:
        result = await download_media_file(query, video=video, max_duration=1800)
    except Exception as e:
        LOGGER.error(f"Download error: {e}")
        return await status.edit_text(f"❌ <b>Download failed:</b> <code>{e}</code>")
    
    if not result or not result.get("path") or not os.path.exists(result["path"]):
        return await status.edit_text("❌ <b>Could not download the file.</b>\nTry a different query or URL.")
    
    file_path = result["path"]
    title = result["title"]
    duration = result["duration"]
    
    try:
        file_size = os.path.getsize(file_path)
        
        if file_size > 2 * 1024 * 1024 * 1024:
            return await status.edit_text("❌ <b>File too large (over 2GB).</b>")
        
        await status.edit_text(f"📤 <b>Uploading:</b> {title[:50]}\n📦 Size: {_fmt_size(file_size)}")
        
        try:
            if video:
                await m.reply_video(video=file_path, caption=f"🎬 <b>{title}</b>\n\n🤖 Downloaded by @{config.BOT_USERNAME.strip('@')}")
            else:
                await m.reply_audio(audio=file_path, title=title[:64], performer="SunloMusic", duration=duration, caption=f"🎵 <b>{title}</b>\n\n🤖 Downloaded by @{config.BOT_USERNAME.strip('@')}")
        except Exception as e:
            LOGGER.error(f"reply_audio/video failed: {e}, trying as document")
            await m.reply_document(document=file_path, caption=f"🎵 <b>{title}</b>\n\n🤖 Downloaded by @{config.BOT_USERNAME.strip('@')}")
        
        if not is_priv:
            await spend_coins(user_id, DOWNLOAD_COST)
            await status.delete()
            await m.reply_text(f"✅ <b>Done!</b> ({DOWNLOAD_COST} coins deducted)")
        else:
            await status.delete()
    except Exception as e:
        LOGGER.error(f"Upload error: {e}")
        await status.edit_text(f"❌ <b>Upload failed:</b> <code>{str(e)[:200]}</code>")
    finally:
        cleanup_download(file_path)

from pyrogram.types import Message
from SunloMusic.bot import bot
from SunloMusic.decorators import catch_errors

@bot.on_message(filters.command("ghost") & filters.group)
@catch_errors
async def ghost_cmd(_, m: Message):
    try:
        # Send ephemeral message
        msg = await bot.send_message(
            chat_id=m.chat.id,
            text="👻 This message will disappear and edit in 3 seconds...",
            ephemeral_message_parameters={"receiver_user_id": m.from_user.id}
        )
        
        await asyncio.sleep(3)
        
        # Edit the ephemeral message
        await bot.edit_ephemeral_message_text(
            chat_id=m.chat.id,
            message_id=msg.id,
            text="✨ <b>Surprise! I got edited.</b>",
            ephemeral_message_parameters={"receiver_user_id": m.from_user.id}
        )
    except Exception as e:
        await m.reply_text(f"Ephemeral not supported: {e}")

# Set Ephemeral Commands (BotCommand is_ephemeral)
@bot.on_message(filters.command("setephemeralcmds") & filters.private)
@catch_errors
async def set_ephemeral_cmds(_, m: Message):
    try:
        from pyrogram.types import BotCommand
        cmds = [
            BotCommand("play", "Play a song (ephemeral)", is_ephemeral=True),
            BotCommand("help", "Show help", is_ephemeral=False)
        ]
        await bot.set_bot_commands(cmds)
        await m.reply_text("✅ Ephemeral commands set!")
    except Exception as e:
        await m.reply_text(f"Failed to set ephemeral commands (API limitation): {e}")

from pyrogram import filters
from pyrogram.types import Message, InlineKeyboardMarkup, InlineKeyboardButton, CallbackQuery

    add_favorite, remove_favorite, get_favorites, is_favorite
)

@bot.on_message(filters.command("like"))
@catch_errors
async def like_cmd(_, m: Message):
    # Works in group (uses current song) OR reply to any message with link
    q = get_queue(m.chat.id) if m.chat.type.name != "PRIVATE" else None
    
    if not q or not q.now_playing:
        return await m.reply_text(
            "❌ <b>Nothing is playing to like.</b>\n\n"
            "💡 <i>Tip: Reply to any song link with /like to save it!</i>"
        )
    
    np = q.now_playing
    if await is_favorite(m.from_user.id, np.url):
        return await m.reply_text("❤️ <b>Already in your favorites!</b>")
    
    await add_favorite(m.from_user.id, {
        "title": np.title,
        "url": np.url,
        "duration": np.duration,
        "thumbnail": np.thumbnail,
        "video": np.video,
    })
    
    await m.reply_text(
        f"❤️ <b>Added to favorites!</b>\n\n"
        f"🎵 {np.title}\n"
        f"━━━━━━━━━━━━━\n"
        f"View all: <code>/favorites</code>"
    )

@bot.on_message(filters.command("favorites"))
@catch_errors
async def favorites_cmd(_, m: Message):
    favs = await get_favorites(m.from_user.id, limit=30)
    if not favs:
        return await m.reply_text(
            "📭 <b>No favorites yet!</b>\n\n"
            "❤️ Use <code>/like</code> while a song is playing"
        )
    
    text = f"❤️ <b>Your Favorites ({len(favs)})</b>\n━━━━━━━━━━━━━\n"
    buttons = []
    for i, fav in enumerate(favs[:15], 1):
        dur = format_duration(fav.get("duration", 0))
        title = fav.get("title", "Unknown")[:40]
        text += f"{i}. {title} <i>({dur})</i>\n"
        buttons.append([InlineKeyboardButton(
            f"▶️ {title[:30]}",
            callback_data=f"favplay_{i-1}"
        )])
    
    buttons.append([InlineKeyboardButton("🗑 Clear All", callback_data="fav_clear")])
    await m.reply_text(text, reply_markup=InlineKeyboardMarkup(buttons))

@bot.on_message(filters.command("playfav") & filters.group)
@catch_errors
async def playfav_cmd(_, m: Message):
    favs = await get_favorites(m.from_user.id, limit=50)
    if not favs:
        return await m.reply_text("📭 <b>No favorites to play.</b>\n\nFirst /like some songs!")
    
    q = get_queue(m.chat.id)
    from SunloMusic.queues import Track
    added = 0
    for fav in favs:
        track = Track(
            fav["title"], fav["url"], fav.get("duration", 0),
            fav.get("thumbnail", ""),
            m.from_user.first_name, m.from_user.id,
            fav.get("video", False)
        )
        if q.now_playing:
            q.queue.append(track)
        else:
            q.queue.append(track)
        added += 1
    
    await m.reply_text(
        f"✅ <b>Added {added} favorite songs to queue!</b>\n\n"
        f"Use <code>/play</code> with any song to start if not playing."
    )

@bot.on_callback_query(filters.regex("^favplay_"))
@catch_errors
async def fav_play_cb(_, cq: CallbackQuery):
    try: idx = int(cq.data.replace("favplay_", ""))
    except: return await cq.answer("Invalid", show_alert=True)
    
    favs = await get_favorites(cq.from_user.id, limit=30)
    if idx >= len(favs):
        return await cq.answer("Not found", show_alert=True)
    
    fav = favs[idx]
    await cq.answer(f"▶️ Sending /play...", show_alert=False)
    try:
        await cq.message.reply_text(f"/play {fav['url']}")
    except: pass

@bot.on_callback_query(filters.regex("^fav_clear$"))
@catch_errors
async def fav_clear_cb(_, cq: CallbackQuery):
    favs = await get_favorites(cq.from_user.id, limit=100)
    for fav in favs:
        await remove_favorite(cq.from_user.id, fav["url"])
    await cq.answer("🗑 All favorites cleared!", show_alert=True)
    try: await cq.message.delete()
    except: pass

import asyncio, random
from pyrogram import filters
from pyrogram.types import Message, InlineKeyboardMarkup, InlineKeyboardButton, CallbackQuery
from SunloMusic.bot import bot
from SunloMusic.decorators import catch_errors

active_battles = {}

@bot.on_message(filters.command("battle") & filters.group)
@catch_errors
async def battle_cmd(_, m: Message):
    if len(m.command) < 2 or "vs" not in m.text.lower():
        return await m.reply_text("❌ <b>Usage:</b> <code>/battle song1 vs song2</code>\nExample: <code>/battle Kesariya vs Tum Hi Ho</code>")
    
    query = m.text.split(None, 1)[1]
    parts = query.lower().split(" vs ")
    if len(parts) != 2:
        return await m.reply_text("❌ <b>Use 'vs' to separate songs.</b>")
    
    song1, song2 = parts[0].strip(), parts[1].strip()
    status = await m.reply_text("⚔️ <b>Preparing Battle...</b>\nFetching both tracks...")
    
    info1 = await extract_audio(song1)
    info2 = await extract_audio(song2)
    
    if not info1 or not info2:
        return await status.edit_text("❌ <b>Could not find one or both songs.</b>")
    
    active_battles[m.chat.id] = {
        "song1": info1, "song2": info2,
        "votes1": 0, "votes2": 0,
        "voters": set()
    }
    
    text = (
        "⚔️ <b>MUSIC BATTLE!</b> ⚔️\n"
        "━━━━━━━━━━━━━━━━━━\n"
        f"🅰️ <b>{info1['title'][:30]}</b>\n"
        "🆚\n"
        f"🅱️ <b>{info2['title'][:30]}</b>\n"
        "━━━━━━━━━━━━━━━━━━\n"
        "👇 <b>Vote for your favorite!</b>\n"
        "<i>(Song snippets will play alternately)</i>"
    )
    
    await status.edit_text(text, reply_markup=InlineKeyboardMarkup([
        [InlineKeyboardButton(f"🅰️ Vote {info1['title'][:15]}", callback_data="vote_1"), InlineKeyboardButton(f"🅱️ Vote {info2['title'][:15]}", callback_data="vote_2")]
    ]))
    
    # Play 15 seconds of Song 1
    await play_stream(m.chat.id, info1["url"], False, seek_time=0)
    await asyncio.sleep(15)
    
    # Play 15 seconds of Song 2
    await play_stream(m.chat.id, info2["url"], False, seek_time=0)
    await asyncio.sleep(15)
    
    try: await status.delete()
    except: pass
    
    if m.chat.id in active_battles:
        data = active_battles.pop(m.chat.id)
        v1, v2 = data["votes1"], data["votes2"]
        winner = info1 if v1 > v2 else info2 if v2 > v1 else None
        
        if winner:
            await m.reply_text(f"🏆 <b>WINNER:</b> {winner['title']}\nWith {max(v1, v2)} votes!")
            track = Track(winner["title"], winner["url"], winner["duration"], winner["thumbnail"], m.from_user.first_name, m.from_user.id, False, video_id=winner.get("video_id", ""))
            q = get_queue(m.chat.id)
            q.now_playing = track
            q.start_time = asyncio.get_event_loop().time()
            await play_stream(m.chat.id, track.url, False)
        else:
            await m.reply_text("🤝 <b>It's a TIE!</b> No winner.")

@bot.on_callback_query(filters.regex(r"^vote_(1|2)$"))
@catch_errors
async def vote_cb(_, cq: CallbackQuery):
    chat_id = cq.message.chat.id
    if chat_id not in active_battles:
        return await cq.answer("Battle has ended!", show_alert=True)
    
    data = active_battles[chat_id]
    if cq.from_user.id in data["voters"]:
        return await cq.answer("You already voted!", show_alert=True)
    
    choice = cq.data.split("_")[1]
    if choice == "1": data["votes1"] += 1
    else: data["votes2"] += 1
    
    data["voters"].add(cq.from_user.id)
    await cq.answer("✅ Vote counted!")

from pyrogram import filters, enums
from pyrogram.types import Message, InlineQueryResultArticle, InputTextMessageContent
from SunloMusic.bot import bot
from SunloMusic.decorators import catch_errors
from SunloMusic.config import config
from SunloMusic.ytdlp import extract_audio
from SunloMusic.logger import LOGGER

@bot.on_guest_message()
@catch_errors
async def guest_handler(_, m: Message):
    query = m.text or ""
    if not query: return
    
    if query.lower().startswith("play "):
        song = query[5:].strip()
        if not song: return
        
        info = await extract_audio(song)
        if info:
            await bot.answer_guest_query(
                m.guest_query_id,
                result=InlineQueryResultArticle(
                    title=f"🎵 {info['title'][:60]}",
                    description=f"⏱ {info['duration']}s • Click to play in VC",
                    thumb_url=info.get("thumbnail"),
                    input_message_content=InputTextMessageContent(
                        message_text=f"/play {info['url']}"
                    )
                )
            )
        else:
            await bot.answer_guest_query(
                m.guest_query_id,
                result=InlineQueryResultArticle(
                    title="❌ Song not found",
                    description="Try a different search",
                    input_message_content=InputTextMessageContent(
                        message_text="Could not find the song."
                    )
                )
            )

async def _safe_edit(cq: CallbackQuery, text: str, reply_markup=None):
    msg = cq.message
    is_media = bool(msg.photo or msg.video or msg.document or msg.animation)
    try:
        if is_media: await msg.edit_caption(caption=text, reply_markup=reply_markup)
        else: await msg.edit_text(text=text, reply_markup=reply_markup)
    except Exception:
        try: await msg.reply_text(text, reply_markup=reply_markup)
        except Exception: pass

async def get_movie_ddl_links(movie_name: str, imdb_id: str = None):
    # 1. Torrentio API (Needs IMDb ID)
    if imdb_id:
        try:
            url = f"https://torrentio.strem.fun/stream/movie/{imdb_id}.json"
            async with aiohttp.ClientSession() as session:
                async with session.get(url, timeout=15) as resp:
                    if resp.status == 200:
                        data = await resp.json()
                        streams = data.get("streams", [])
                        if streams:
                            for s in streams:
                                if "url" in s and s["url"].startswith("magnet:"):
                                    LOGGER.info(f"Torrentio found link for {movie_name} ({imdb_id})")
                                    return {"4k": s["url"], "1080p": s["url"], "720p": s["url"]}
        except Exception as e:
            LOGGER.error(f"Torrentio Error: {e}")

    # 2. PirateBay API (With User-Agent to prevent block)
    try:
        clean_name = movie_name.split(':')[0].split('-')[0].strip()
        query = clean_name.replace(' ', '%20')
        headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"}
        async with aiohttp.ClientSession(headers=headers) as session:
            async with session.get(f"https://apibay.org/q.php?q={query}&cat=200", timeout=15) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    if data and len(data) > 0 and data[0].get("id") != "0":
                        results = sorted([r for r in data if int(r.get("seeders", 0)) > 0], key=lambda x: int(x["seeders"]), reverse=True)
                        if results:
                            best = results[0]
                            info_hash = best["info_hash"]
                            trackers = "&tr=udp%3A%2F%2Ftracker.openbittorrent.com%3A80&tr=udp%3A%2F%2Ftracker.opentrackr.org%3A1337"
                            magnet = f"magnet:?xt=urn:btih:{info_hash}{trackers}"
                            LOGGER.info(f"PirateBay found link for {clean_name}")
                            return {"4k": magnet, "1080p": magnet, "720p": magnet}
    except Exception as e:
        LOGGER.error(f"PirateBay Error: {e}")

    return None

async def search_movie_tmdb(query: str):
    if not config.TMDB_API_KEY: return None
    params = {"api_key": config.TMDB_API_KEY, "query": query, "language": "en-US", "page": 1}
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get("https://api.themoviedb.org/3/search/movie", params=params, timeout=15) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    results = data.get("results", [])
                    if results: return results[0]

                if resp.status == 200:
                    data = await resp.json()
                    return data.get("imdb_id", "")
    except: pass
    return ""

def _search_sync(query: str):
    try:
        with yt_dlp.YoutubeDL({"quiet": True, "extract_flat": True, "noplaylist": True}) as ydl:
            info = ydl.extract_info(f"ytsearch5:{query} audio", download=False)
            results = []
            if info and "entries" in info:
                for e in info["entries"]:
                    if e:
                        results.append({
                            "title": e.get("title", "Unknown"),
                            "url": e.get("url"),
                            "duration": e.get("duration", 0),
                            "id": e.get("id")
                        })
            return results
    except Exception as e:
        LOGGER.error(f"Search error: {e}")
        return []

    if not results:
        return await status.edit_text("❌ <b>No results found.</b>")
        
    buttons = []

STICKER_DIR = "cache/stickers"

# 1. Text to Animated Sticker (Neon Glow)
def create_sticker_image(text: str, output_path: str):
    img = Image.new("RGBA", (512, 512), (0, 0, 0, 0))
    draw = ImageDraw.Draw(img)
    try: font = ImageFont.truetype("assets/fonts/bold.ttf", 70)
    except: font = ImageFont.load_default()

    bbox = draw.textbbox((0, 0), text, font=font)
    x, y = (512 - (bbox[2] - bbox[0])) / 2, (512 - (bbox[3] - bbox[1])) / 2

    glow_layer = Image.new("RGBA", (512, 512), (0, 0, 0, 0))
    glow_draw = ImageDraw.Draw(glow_layer)
    glow_draw.text((x, y), text, font=font, fill=(0, 255, 255, 255))
    glow_layer = glow_layer.filter(ImageFilter.GaussianBlur(15))
    
    img = Image.alpha_composite(img, glow_layer)
    draw = ImageDraw.Draw(img)
    draw.text((x, y), text, font=font, fill=(255, 255, 255, 255), stroke_width=3, stroke_fill=(0, 0, 0, 255))
    img.save(output_path, "PNG")
    return output_path

def convert_to_webm(image_path: str, output_path: str, is_zoom: bool = False):
    if is_zoom:
        # ✅ Photo se Live Sticker (Zoom-in effect)
        vf_filter = "scale=512:512:force_original_aspect_ratio=decrease,pad=512:512:(ow-iw)/2:(oh-ih)/2,zoompan=z='min(zoom+0.0015,1.5)':d=60:s=512x512,format=yuva420p"
    else:
        # ✅ Text sticker filter
        vf_filter = "scale=512:512,format=yuva420p,zoompan=z='min(zoom+0.0015,1.5)':d=60:s=512x512"

    command = [
        "ffmpeg", "-y", "-loop", "1", "-i", image_path,
        "-vf", vf_filter,
        "-c:v", "libvpx-vp9", "-b:v", "256k", "-crf", "35",
        "-t", "3.0", "-an", output_path
    ]
    try:
        subprocess.run(command, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=True)
        return True
    except Exception as e:
        LOGGER.error(f"FFmpeg Error: {e}")
        return False

def generate_styles(name: str) -> list:
    styles = set()
    stylized_names = {f_name: func(name) for f_name, func in FONTS.items()}
    
    # 1. Pure Emoji + Font + Emoji
    for s_name in stylized_names.values():
        for emoji in EMOJIS:
            styles.add(f"{emoji} {s_name} {emoji}")
            
    # 2. Apply all frames to all fonts
    for s_name in stylized_names.values():
        for frame in FRAMES:
            styles.add(frame.format(s_name))

    # 3. Double Emoji + Frame
    for s_name in stylized_names.values():
        for i in range(0, len(FRAMES), 3):
            frame = FRAMES[i]
            emo1 = random.choice(EMOJIS)
            emo2 = random.choice(EMOJIS)
            styles.add(f"{emo1} {frame.format(s_name)} {emo2}")

    # 4. Special Strikethrough / Underline / Zalgo
    for s_name in stylized_names.values():
        styles.add("".join([c + "\u0336" for c in s_name]))
        styles.add("".join([c + "\u0332" for c in s_name]))
        
    # 5. Mix and Match until 600+ styles
    all_frames = FRAMES + [""]
    all_names = list(stylized_names.values())
    
    while len(styles) < 600:
        frame = random.choice(all_frames)
        nm = random.choice(all_names)
        emo1 = random.choice(EMOJIS + [""])
        emo2 = random.choice(EMOJIS + [""])
        sp = random.choice([" ", "  ", "", " "])
        
        base = frame.format(nm) if frame else nm
        styles.add(f"{emo1}{sp}{base}{sp}{emo2}".strip())

    style_list = list(styles)
    random.shuffle(style_list)
    return style_list

async def send_style_page(message_obj, user_id: int, page: int, is_edit: bool = True):
    data = STYLE_CACHE.get(user_id)
    if not data:
        return
        
    styles = data["styles"]
    name = data["name"]
    
    items_per_page = 20
    total_pages = math.ceil(len(styles) / items_per_page)
    page = max(1, min(page, total_pages))
    
    start_idx = (page - 1) * items_per_page
    end_idx = start_idx + items_per_page
    page_styles = styles[start_idx:end_idx]
    
    # Premium Card Format
    text = (
        f"✨ <b>『 ᴘʀᴇᴍɪᴜᴍ ꜱᴛʏʟᴇꜱ 』</b> ✨\n"
        f"▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬\n"
        f"👤 <b>Name:</b> <code>{name}</code>\n"
        f"🎨 <b>Total Designs:</b> <code>{len(styles)}</code>\n"
        f"📄 <b>Page:</b> <code>{page}/{total_pages}</code>\n"
        f"▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬\n"
        f"👇 <i>Tap on any design to copy it!</i>\n\n"
    )
    
    for i, style in enumerate(page_styles, 1):
        text += f"{start_idx + i}. <code>{style}</code>\n"
        
    # Pagination Buttons
    buttons = []
    nav_row = []
    if page > 1:
        nav_row.append(InlineKeyboardButton("⬅️ ᴘʀᴇᴠ", callback_data=f"style_page_{user_id}_{page-1}", style=ButtonStyle.PRIMARY))
    nav_row.append(InlineKeyboardButton(f"📄 {page}/{total_pages}", callback_data="style_ignore", style=ButtonStyle.SUCCESS))
    if page < total_pages:
        nav_row.append(InlineKeyboardButton("ɴᴇxᴛ ➡️", callback_data=f"style_page_{user_id}_{page+1}", style=ButtonStyle.PRIMARY))
        
    buttons.append(nav_row)
    buttons.append([InlineKeyboardButton("❌ ᴄʟᴏꜱᴇ", callback_data=f"style_close_{user_id}", style=ButtonStyle.DANGER)])
    
    reply_markup = InlineKeyboardMarkup(buttons)
    
    if is_edit:
        try:
            await message_obj.edit_text(text, reply_markup=reply_markup, link_preview_options=LinkPreviewOptions(is_disabled=True))
        except:
            pass
    else:
        try:
            await message_obj.edit_text(text, reply_markup=reply_markup, link_preview_options=LinkPreviewOptions(is_disabled=True))
        except:
            await message_obj.delete()
            await bot.send_message(message_obj.chat.id, text, reply_markup=reply_markup, link_preview_options=LinkPreviewOptions(is_disabled=True))

TTS_DIR = "cache/tts"

async def speak_in_vc(chat_id: int, text: str):
    if not await get_group_setting(chat_id, "tts", False): return
    q = get_queue(chat_id)
    if not call: return
    try:
        voice = "hi-IN-SwaraNeural" 
        output_file = f"{TTS_DIR}/voice_{chat_id}_{int(time.time())}.mp3"
        communicate = edge_tts.Communicate(text, voice)
        await communicate.save(output_file)
        
        if os.path.exists(output_file) and os.path.getsize(output_file) > 0:
            q.is_playing_tts = True
            await call.play(chat_id, MediaStream(media_path=output_file, audio_parameters=AudioQuality.HIGH))
            await asyncio.sleep(len(text) * 0.1 + 1.5) 
            q.is_playing_tts = False
    except Exception as e: LOGGER.error(f"TTS Error: {e}")
